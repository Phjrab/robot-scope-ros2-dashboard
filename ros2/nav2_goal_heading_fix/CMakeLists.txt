cmake_minimum_required(VERSION 3.16)
project(robot_scope_heading_reproduction)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(dependencies rclcpp geometry_msgs nav2_costmap_2d pluginlib nav_msgs
    nav2_controller nav2_util nav2_core tf2 angles)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
find_package(ament_cmake_gtest REQUIRED)
find_package(nav2_rotation_shim_controller REQUIRED)
enable_testing()
foreach(variant original tf_only fixed installed)
  if(variant STREQUAL "original")
    set(source_dir "${UPSTREAM_DIR}/nav2_rotation_shim_controller")
  elseif(variant STREQUAL "tf_only")
    set(source_dir "${TF_ONLY_DIR}/nav2_rotation_shim_controller")
  else()
    set(source_dir "${PATCHED_DIR}/nav2_rotation_shim_controller")
  endif()
  ament_add_gtest(test_${variant} test_heading.cpp TIMEOUT 45)
  if(variant STREQUAL "installed")
    ament_target_dependencies(test_${variant} nav2_rotation_shim_controller)
  else()
    add_library(shim_${variant} SHARED "${source_dir}/src/nav2_rotation_shim_controller.cpp")
    target_include_directories(shim_${variant} BEFORE PUBLIC "${source_dir}/include")
    ament_target_dependencies(shim_${variant} ${dependencies})
    target_link_libraries(test_${variant} shim_${variant})
  endif()
  ament_target_dependencies(test_${variant} ${dependencies})
  if(variant STREQUAL "fixed" OR variant STREQUAL "tf_only")
    target_compile_definitions(test_${variant} PRIVATE ROBOT_SCOPE_FIXED=1)
  endif()
  if(variant STREQUAL "fixed")
    target_compile_definitions(test_${variant} PRIVATE ROBOT_SCOPE_COLLISION_FIXED=1)
  endif()
endforeach()
if(CANDIDATE_LIBRARY)
  ament_add_gtest(test_production_library test_heading.cpp TIMEOUT 45)
  target_include_directories(test_production_library BEFORE PRIVATE
    "${PATCHED_DIR}/nav2_rotation_shim_controller/include")
  target_link_libraries(test_production_library "${CANDIDATE_LIBRARY}")
  ament_target_dependencies(test_production_library ${dependencies})
  target_compile_definitions(test_production_library PRIVATE ROBOT_SCOPE_FIXED=1 ROBOT_SCOPE_COLLISION_FIXED=1)
  get_filename_component(candidate_lib_dir "${CANDIDATE_LIBRARY}" DIRECTORY)
  set_tests_properties(test_production_library PROPERTIES ENVIRONMENT
    "LD_LIBRARY_PATH=${candidate_lib_dir}:$ENV{LD_LIBRARY_PATH};ROBOT_SCOPE_EXPECTED_SHIM_LIBRARY=${CANDIDATE_LIBRARY}")
endif()
